# useStore

```ts
useStore(): Store
```

Returns the exact Store from the nearest `Provider`. It throws when there is no
Provider; there is no global fallback or string-ID registry.

```tsx
import { useCallback } from "react"
import { useStore } from "valdres-react"

function ResetProfileButton() {
    const store = useStore()
    const resetProfile = useCallback(() => {
        store.txn(transaction => {
            transaction.reset(nameAtom)
            transaction.reset(emailAtom)
        })
    }, [store])

    return <button onClick={resetProfile}>Reset profile</button>
}
```
