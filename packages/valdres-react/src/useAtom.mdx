# useAtom

```ts
useAtom<Value>(atom: Atom<Value>, store?: Store):
    readonly [Value, (value: Value) => void]
```

Returns the Atom value and an exact-value setter. The setter does not interpret
functions as updater callbacks. Use `useUpdateAtom` for updater semantics.

```tsx
function Counter() {
    const [count, setCount] = useAtom(countAtom)
    const updateCount = useUpdateAtom(countAtom)

    return (
        <>
            <span>{count}</span>
            <button onClick={() => setCount(0)}>Clear</button>
            <button onClick={() => updateCount(value => value + 1)}>+</button>
        </>
    )
}
```
